// ───────────────────────── Engine ─────────────────────────
newEngine() =>
    Engine.new(array.new_float(), array.new_float(), array.new_float(), array.new_float(), array.new_float(), array.new_int(), array.new_int(), array.new_float(), array.new_float(), array.new_float(), array.new_int(), array.new_float(), array.new_int())

method fa(Engine e, array<float> a, int i) =>
    a.get(i - e.base)

method ia(Engine e, array<int> a, int i) =>
    a.get(i - e.base)

// Runs the six impulse checks on bars s..en. leg.fail = "" when all pass, else the failed checks.
method validate(Engine e, int dir, int s, int en, float lo, float hi, Cfg cfg) =>
    Leg res = na
    int bars = en - s + 1
    if s - 1 >= e.base and bars >= 2
        float atrRef = e.fa(e.atr, s - 1)
        float vaRef  = e.fa(e.va, s - 1)
        float path = 0.0
        float vSum = 0.0
        int   nDir = 0
        for k = s to en
            float ok = e.fa(e.o, k)
            float ck = e.fa(e.c, k)
            path += e.fa(e.h, k) - e.fa(e.l, k)
            vSum += nz(e.fa(e.v, k))
            if (dir == 1 and ck > ok) or (dir == -1 and ck < ok)
                nDir += 1
        float net    = dir == 1 ? e.fa(e.c, en) - e.fa(e.o, s) : e.fa(e.o, s) - e.fa(e.c, en)
        float eff    = path > 0 ? net / path : 0.0
        float sizeX  = atrRef > 0 ? (hi - lo) / atrRef : 0.0
        float volX   = vaRef > 0 ? (vSum / bars) / vaRef : 0.0
        // the origin / extreme candle may close against the leg (wick reversal): not counted
        bool  firstCtr = dir == 1 ? e.fa(e.c, s) <= e.fa(e.o, s) : e.fa(e.c, s) >= e.fa(e.o, s)
        bool  lastCtr  = dir == 1 ? e.fa(e.c, en) <= e.fa(e.o, en) : e.fa(e.c, en) >= e.fa(e.o, en)
        int   nDen     = bars - (firstCtr ? 1 : 0) - (lastCtr ? 1 : 0)
        float dirPct   = float(nDir) / math.max(nDen, 1)
        bool  bos    = not cfg.reqBos
        if cfg.reqBos
            array<float> lv = dir == 1 ? e.phL : e.plL
            array<int>   ix = dir == 1 ? e.phI : e.plI
            int np = ix.size()
            if np > 0
                for k = np - 1 to 0
                    if ix.get(k) < s
                        float lvl = lv.get(k)
                        bos := dir == 1 ? hi > lvl : lo < lvl
                        break
        string fail = ""
        if bars < cfg.minBars or bars > cfg.maxBars
            fail += " bars"
        if not bos
            fail += " BOS"
        if not (sizeX >= cfg.minAtr)
            fail += " size"
        if not (eff >= cfg.minEff)
            fail += " eff"
        if not (volX >= cfg.minVol)
            fail += " vol"
        if not (dirPct >= cfg.minDir)
            fail += " dir"
        res := Leg.new(dir, e.ia(e.t, s), e.ia(e.tc, en), lo, hi, bars, sizeX, eff, volX, dirPct, fail)
    res

// Feeds one closed bar. Returns candidate legs finalized on this bar (after the pullback), passed or not.
method feed(Engine e, float o, float h, float l, float c, float v, int t, int tc, float atr, float va, Cfg cfg) =>
    array<Leg> out = array.new<Leg>()
    e.o.push(o)
    e.h.push(h)
    e.l.push(l)
    e.c.push(c)
    e.v.push(v)
    e.t.push(t)
    e.tc.push(tc)
    e.atr.push(atr)
    e.va.push(va)
    e.n := e.n + 1
    int i = e.n - 1
    if e.o.size() > 800
        e.o.shift()
        e.h.shift()
        e.l.shift()
        e.c.shift()
        e.v.shift()
        e.t.shift()
        e.tc.shift()
        e.atr.shift()
        e.va.shift()
        e.base := e.base + 1

    // swing pivots (confirmed pivLen bars later)
    int L = cfg.pivLen
    int p = i - L
    if p - L >= e.base
        float ph = e.fa(e.h, p)
        float pl = e.fa(e.l, p)
        bool isH = true
        bool isL = true
        for k = p - L to p + L
            if k != p
                if e.fa(e.h, k) >= ph
                    isH := false
                if e.fa(e.l, k) <= pl
                    isL := false
        if isH
            e.phL.push(ph)
            e.phI.push(p)
            e.lastSH   := ph
            e.shBroken := false
        if isL
            e.plL.push(pl)
            e.plI.push(p)
            e.lastSL   := pl
            e.slBroken := false
        if e.phL.size() > 60
            e.phL.shift()
            e.phI.shift()
        if e.plL.size() > 60
            e.plL.shift()
            e.plI.shift()

    // structure breaks -> bias (close-based)
    if not e.shBroken and c > e.lastSH
        e.shBroken := true
        e.choch    := e.trend == -1
        e.trend    := 1
        e.lastBrk  := i
    if not e.slBroken and c < e.lastSL
        e.slBroken := true
        e.choch    := e.trend == 1
        e.trend    := -1
        e.lastBrk  := i

    // bull tracker: origin low -> extreme high, finalized on pullback
    if not e.bOn
        e.bOn := true
        e.bO  := i
        e.bLo := l
        e.bE  := i
        e.bHi := h
    else
        float rng = e.bHi - e.bLo
        // a bar that makes a new high extends the leg, even if its wick dips (pullback only after the extreme)
        if h <= e.bHi and e.bE > e.bO and rng > 0 and e.bHi - l >= cfg.pb * rng and (na(atr) or e.bHi - l >= cfg.pbAtr * atr)
            Leg lg = e.validate(1, e.bO, e.bE, e.bLo, e.bHi, cfg)
            if not na(lg)
                out.push(lg)
            e.bO  := i
            e.bLo := l
            e.bE  := i
            e.bHi := h
        else if l < e.bLo
            e.bO  := i
            e.bLo := l
            e.bE  := i
            e.bHi := h
        else if h > e.bHi
            e.bHi := h
            e.bE  := i
            if e.bE - e.bO + 1 > cfg.maxBars
                int s0 = e.bE - cfg.maxBars + 1
                float mn = e.fa(e.l, s0)
                int mi = s0
                for k = s0 + 1 to e.bE
                    if e.fa(e.l, k) < mn
                        mn := e.fa(e.l, k)
                        mi := k
                e.bO  := mi
                e.bLo := mn

    // bear tracker: origin high -> extreme low
    if not e.sOn
        e.sOn := true
        e.sO  := i
        e.sHi := h
        e.sE  := i
        e.sLo := l
    else
        float rng = e.sHi - e.sLo
        if l >= e.sLo and e.sE > e.sO and rng > 0 and h - e.sLo >= cfg.pb * rng and (na(atr) or h - e.sLo >= cfg.pbAtr * atr)
            Leg lg = e.validate(-1, e.sO, e.sE, e.sLo, e.sHi, cfg)
            if not na(lg)
                out.push(lg)
            e.sO  := i
            e.sHi := h
            e.sE  := i
            e.sLo := l
        else if h > e.sHi
            e.sO  := i
            e.sHi := h
            e.sE  := i
            e.sLo := l
        else if l < e.sLo
            e.sLo := l
            e.sE  := i
            if e.sE - e.sO + 1 > cfg.maxBars
                int s0 = e.sE - cfg.maxBars + 1
                float mx = e.fa(e.h, s0)
                int mi = s0
                for k = s0 + 1 to e.sE
                    if e.fa(e.h, k) > mx
                        mx := e.fa(e.h, k)
                        mi := k
                e.sO  := mi
                e.sHi := mx
    out

// ───────────────────────── Volume profile / LVN ─────────────────────────
// Profile of [t0, t1) between lo..hi from the intrabar buffer. Returns [top, bot, top, bot, ...] for the
// deepest LVNs that sit between two HVNs inside the inner (1 - 2*edgePct) of the leg.
f_lvn(int t0, int t1, float lo, float hi) =>
    array<float> res = array.new_float()
    float step = (hi - lo) / rows
    int sz = bT.size()
    if step > 0 and sz > 0
        array<float> prof = array.new_float(rows, 0.0)
        array<float> tp   = array.new_float(rows, 0.0)
        float tSum = 0.0
        int k = math.max(0, bT.binary_search_leftmost(t0))
        while k < sz
            int tk = bT.get(k)
            if tk >= t1
                break
            if tk >= t0
                float bh = math.min(bH.get(k), hi)
                float bl = math.max(bL.get(k), lo)
                float vv = bV.get(k)
                if bh >= bl and vv > 0
                    int r0 = math.max(0, math.min(rows - 1, math.floor((bl - lo) / step)))
                    int r1 = math.max(0, math.min(rows - 1, math.floor((bh - lo) / step)))
                    float share = vv / (r1 - r0 + 1)
                    for r = r0 to r1
                        prof.set(r, prof.get(r) + share)
                        tp.set(r, tp.get(r) + 1)
                    tSum += r1 - r0 + 1
            k += 1

        // 3-row smoothing
        array<float> sm = array.new_float(rows, 0.0)
        for r = 0 to rows - 1
            float sv = prof.get(r)
            int cnt = 1
            if r > 0
                sv += prof.get(r - 1)
                cnt += 1
            if r < rows - 1
                sv += prof.get(r + 1)
                cnt += 1
            sm.set(r, sv / cnt)

        // HVNs = highest row within +-peakW rows
        array<int> pk = array.new_int()
        for r = 0 to rows - 1
            float cur = sm.get(r)
            bool isPk = cur > 0
            for q = math.max(0, r - peakW) to math.min(rows - 1, r + peakW)
                if (q < r and sm.get(q) > cur) or (q > r and sm.get(q) >= cur)
                    isPk := false
            if isPk
                pk.push(r)

        // valley between each two adjacent HVNs, inside the inner zone
        int rIn0 = math.ceil(rows * edgePct)
        int rIn1 = rows - 1 - rIn0
        array<float> cR = array.new_float()
        array<float> cT = array.new_float()
        array<float> cB = array.new_float()
        if pk.size() >= 2
            for j = 0 to pk.size() - 2
                int p1 = pk.get(j)
                int p2 = pk.get(j + 1)
                if p2 - p1 >= 2
                    int m = p1 + 1
                    for r = p1 + 1 to p2 - 1
                        if sm.get(r) < sm.get(m)
                            m := r
                    float refV  = math.min(sm.get(p1), sm.get(p2))
                    float ratio = refV > 0 ? sm.get(m) / refV : 1.0
                    if m >= rIn0 and m <= rIn1 and ratio <= maxRatio
                        // widen the zone over neighbouring low-volume rows
                        float vMin = sm.get(m)
                        float lim  = vMin + 0.30 * (refV - vMin)
                        int z0 = m
                        int z1 = m
                        while z0 - 1 > p1 and z0 - 1 >= rIn0 and sm.get(z0 - 1) <= lim
                            z0 -= 1
                        while z1 + 1 < p2 and z1 + 1 <= rIn1 and sm.get(z1 + 1) <= lim
                            z1 += 1
                        // time at price: intrabars at the zone middle row vs the leg average per row
                        float tX = tSum > 0 ? tp.get(math.min(z1, math.floor((z0 + z1 + 1) / 2))) / (tSum / rows) : 1.0
                        if maxTime == 0 or tX <= maxTime
                            cR.push(ratio)
                            cT.push(lo + (z1 + 1) * step)
                            cB.push(lo + z0 * step)

        // keep the deepest maxLvn
        while cR.size() > maxLvn
            int ix = cR.indexof(cR.max())
            cR.remove(ix)
            cT.remove(ix)
            cB.remove(ix)
        if cR.size() > 0
            for j = 0 to cR.size() - 1
                res.push(cT.get(j))
                res.push(cB.get(j))
    res

// ───────────────────────── Zones ─────────────────────────
f_stName(int s) =>
    s == 0 ? "FRESH" : s == 1 ? "TESTED" : s == 2 ? "WEAK" : s == 3 ? "FILLED" : s == 4 ? "BROKEN" : "LEG INVALID"

f_aligned(Zone z) =>
    not biasFilt or biasDir == 0 or z.dir == biasDir

f_vis(Zone z) =>
    bool dead = z.status >= 3
    (not onlyFresh or z.status == 0) and (not dead or keepDead) and (f_aligned(z) or showCtr) and (z.htf ? showHtf : showLtf) and (z.htf or not onlyNest or z.nested)

f_draw(Zone z) =>
    bool dead = z.status >= 3
    color bg = dead ? cDead : not f_aligned(z) ? cCtr : z.status == 0 ? (z.dir == 1 ? cBull : cBear) : z.status == 1 ? cTest : cWeak
    z.bx.set_right(dead ? z.tDead : time)
    if f_vis(z)
        string tf = z.htf ? htfLbl : chartLbl
        z.bx.set_bgcolor(bg)
        z.bx.set_border_color(color.new(bg, 20))
        z.bx.set_border_width(z.nested or z.htf ? 2 : 1)
        z.bx.set_text(tf + " LVN" + (z.nested ? " • NESTED" : "") + " • " + f_stName(z.status))
    else
        z.bx.set_bgcolor(color.new(color.black, 100))
        z.bx.set_border_color(color.new(color.black, 100))
        z.bx.set_text("")
    true

// Did price come back to the zone after the impulse first left it fully behind (inside the leg or after it)?
f_seen(int t0, int dir, float top, float bot) =>
    bool seen = false
    bool away = false
    int sz = bT.size()
    if sz > 0
        int k = math.max(0, bT.binary_search_leftmost(t0))
        while k < sz and not seen
            if bT.get(k) >= t0
                if away
                    seen := dir == 1 ? bL.get(k) <= top : bH.get(k) >= bot
                else
                    away := dir == 1 ? bL.get(k) > top : bH.get(k) < bot
            k += 1
    seen

f_trim() =>
    while zones.size() > maxZones
        int di = 0
        for k = 0 to zones.size() - 1
            if zones.get(k).status >= 3
                di := k
                break
        Zone z = zones.remove(di)
        z.bx.delete()
    zones.size()

f_stats(Leg lg) =>

// Builds the LVN zone for a finalized leg (+ optional leg line). Returns true if a zone was made.
f_leg(Leg lg, bool isHtf) =>
    array<float> zs = f_lvn(lg.tStart, lg.tEnd, lg.lo, lg.hi)
    int nZ = math.round(zs.size() / 2)
    bool made = nZ > 0
    if made
        for i = 0 to nZ - 1
            float zt = zs.get(i * 2)
            float zb = zs.get(i * 2 + 1)
            Zone z = Zone.new(top = zt, bot = zb, dir = lg.dir, htf = isHtf, origin = lg.dir == 1 ? lg.lo : lg.hi, tLeft = lg.tStart, tCreate = time)
            if f_seen(lg.tStart, lg.dir, zt, zb)
                z.status := 1
            z.bx := box.new(left = lg.tStart, top = zt, right = time, bottom = zb, xloc = xloc.bar_time, border_color = color.new(color.black, 100), bgcolor = color.new(color.black, 100), text = "", text_size = size.tiny, text_color = chart.fg_color, text_halign = text.align_right, text_valign = text.align_top)
            for oz in zones
                if oz.status < 3 and oz.htf != isHtf and oz.dir == z.dir and z.top >= oz.bot and z.bot <= oz.top
                    if isHtf
                        oz.nested := true
                    else
                        z.nested := true
            zones.push(z)
        f_trim()
    if showLegs
        color lc = lg.dir == 1 ? color.teal : color.maroon
        line.new(lg.tStart, lg.dir == 1 ? lg.lo : lg.hi, lg.tEnd, lg.dir == 1 ? lg.hi : lg.lo, xloc = xloc.bar_time, color = lc, width = isHtf ? 2 : 1, style = isHtf ? line.style_solid : line.style_dotted)
        if isHtf
            string txt = htfLbl + " leg  " + f_stats(lg) + (made ? "" : "  (no LVN)")
            label.new(lg.tEnd, lg.dir == 1 ? lg.hi : lg.lo, txt, xloc = xloc.bar_time, style = lg.dir == 1 ? label.style_label_down : label.style_label_up, color = color.new(lc, 30), textcolor = color.white, size = size.tiny)
    made

// Debug: grey line + which checks failed
f_reject(Leg lg, bool isHtf) =>

        line.new(lg.tStart, lg.dir == 1 ? lg.lo : lg.hi, lg.tEnd, lg.dir == 1 ? lg.hi : lg.lo, xloc = xloc.bar_time, color = color.gray, width = isHtf ? 2 : 1, style = line.style_dashed)
        label.new(lg.tEnd, lg.dir == 1 ? lg.hi : lg.lo, "✗ " + tf + ":" + lg.fail + "\n" + str.tostring(lg.bars) + " bars  " + f_stats(lg), xloc = xloc.bar_time, style = lg.dir == 1 ? label.style_label_down : label.style_label_up, color = color.new(color.gray, 30), textcolor = color.white, size = size.tiny)
    true

// Most extreme value of a from index i0 to the end, as [value, index]
f_extreme(array<float> a, int i0, bool wantMax) =>
    float v = a.get(i0)
    int   ix = i0
    for k = i0 to a.size() - 1
        if wantMax ? a.get(k) > v : a.get(k) < v
            v  := a.get(k)
            ix := k
    [v, ix]

// Draws the confirmed swing and names it against the previous swing on the same side
f_swing(bool isHigh, float px, float prevSame, int bar) =>
    if showSwg
        string nm = na(prevSame) ? (isHigh ? "H" : "L") : isHigh ? (px > prevSame ? "HH" : "LH") : (px > prevSame ? "HL" : "LL")
        if not na(lastBar)
            line.new(lastBar, lastPx, bar, px, color = color.new(color.silver, 40))
        label.new(bar, px, nm, style = isHigh ? label.style_label_down : label.style_label_up, color = color.new(color.black, 100), textcolor = color.silver, size = size.small)
    true

// One candle of the reaction timeframe for one zone: touch -> CHoCH beyond the last swing -> BOS beyond a swing made after the CHoCH.
f_react(Zone z, int t, float hi, float lo, float cl, float swH, int tH, float swL, int tL, string tf) =>
    int res = 0
    if z.reStep == 0 and (z.dir == 1 ? lo <= z.top : hi >= z.bot)
        z.reStep := 1
        z.reShow := showRe and f_vis(z)
    float lvl = z.dir == 1 ? swH : swL
    int   tLv = z.dir == 1 ? tH : tL
    if not na(lvl) and (z.reStep == 1 or (z.reStep == 2 and tLv > z.tChoch)) and (z.dir == 1 ? cl > lvl : cl < lvl)
        bool isBos = z.reStep == 2
        z.reStep := isBos ? 3 : 2
        z.tChoch := t
        res := isBos ? 2 : 1
        if z.reShow
            color mc = z.dir == 1 ? color.lime : color.red
            string nm = tf + (isBos ? " BOS" : " CHoCH")
            line.new(tLv, lvl, t, lvl, xloc = xloc.bar_time, color = color.new(mc, isBos ? 0 : 40), style = isBos ? line.style_solid : line.style_dashed)
            label.new(t, lvl, nm, xloc = xloc.bar_time, style = z.dir == 1 ? label.style_label_down : label.style_label_up, color = color.new(mc, isBos ? 10 : 60), textcolor = isBos ? color.black : color.white, size = size.tiny)
            if isBos
                alert(nm + " at " + (z.htf ? htfLbl : chartLbl) + " LVN (" + (z.dir == 1 ? "demand" : "supply") + ") on " + syminfo.ticker + ": " + str.tostring(z.bot, format.mintick) + " – " + str.tostring(z.top, format.mintick), alert.freq_once_per_bar_close)
    res

    if nl > 0
